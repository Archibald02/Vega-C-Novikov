cmake_minimum_required(VERSION 3.20)
project(QuantPriceLive VERSION 6.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
option(BUILD_GUI "Build native Qt desktop application" ON)
option(BUILD_BENCHMARKS "Build Google Benchmark executable" ON)
include(FetchContent)
add_library(quant src/analytics.cpp src/indicators.cpp src/research.cpp src/robustness.cpp)
target_include_directories(quant PUBLIC include)
if(MSVC)
  target_compile_options(quant PRIVATE /W4)
else()
  target_compile_options(quant PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets Concurrent)
  add_library(quote_parser app/quote_parser.cpp)
  target_include_directories(quote_parser PUBLIC app)
  target_link_libraries(quote_parser PUBLIC Qt6::Core)
  add_library(live_widgets app/research_panel.cpp app/research_workbench.cpp app/research_experiments.cpp app/research_journey.cpp app/research_panel.hpp app/json_http.cpp app/market_history.cpp app/market_history.hpp app/visual_chart.cpp app/history_panel.cpp
    app/main_window.cpp app/workspace_navigation.cpp app/main_window.hpp
    app/instrument_catalog.cpp app/instrument_catalog.hpp app/catalog_dialog.cpp app/catalog_dialog.hpp app/resource_monitor.cpp
    app/chart_widget.cpp app/chart_widget.hpp app/market_feed.cpp app/market_feed.hpp)
  set_target_properties(live_widgets PROPERTIES AUTOMOC ON)
  target_link_libraries(live_widgets PUBLIC quant quote_parser Qt6::Widgets Qt6::Network Qt6::WebSockets Qt6::Concurrent)
  if(WIN32)
    target_link_libraries(live_widgets PRIVATE psapi)
  endif()
  qt_add_executable(quant_price_live WIN32 app/main.cpp)
  target_link_libraries(quant_price_live PRIVATE live_widgets)
endif()
if(BUILD_TESTING)
  FetchContent_Declare(doctest GIT_REPOSITORY https://github.com/doctest/doctest.git GIT_TAG v2.4.12 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(doctest)
  add_executable(quant_tests tests/test_analytics.cpp)
  target_link_libraries(quant_tests PRIVATE quant doctest::doctest)
  add_test(NAME quant_unit_tests COMMAND quant_tests)
  add_executable(research_tests tests/test_research.cpp)
  target_link_libraries(research_tests PRIVATE quant doctest::doctest)
  target_compile_definitions(research_tests PRIVATE REFERENCE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/research_reference.tsv")
  add_test(NAME quantitative_and_strategy_tests COMMAND research_tests)
  add_executable(indicator_tests tests/test_indicators.cpp)
  target_link_libraries(indicator_tests PRIVATE quant doctest::doctest)
  add_test(NAME indicators_forecast_and_risk COMMAND indicator_tests)
  if(BUILD_GUI)
    add_executable(network_tests tests/test_network.cpp)
    target_link_libraries(network_tests PRIVATE live_widgets doctest::doctest)
    add_test(NAME network_failures_and_privacy COMMAND network_tests)
    set_tests_properties(network_failures_and_privacy PROPERTIES TIMEOUT 30)
    add_executable(quote_tests tests/test_quotes.cpp)
    target_link_libraries(quote_tests PRIVATE quote_parser doctest::doctest)
    add_test(NAME provider_payload_tests COMMAND quote_tests)
    add_test(NAME gui_smoke_test COMMAND quant_price_live --smoke-test)
    add_test(NAME gui_tools_test COMMAND quant_price_live --tools-test)
    add_test(NAME gui_laboratory_test COMMAND quant_price_live --lab-test)
    add_test(NAME gui_visuals_test COMMAND quant_price_live --visuals-test --view 7 --screenshot "${CMAKE_BINARY_DIR}/gui-visuals.png")
    add_test(NAME gui_privacy_and_lifecycle COMMAND quant_price_live --privacy-test)
    add_test(NAME bounded_history_stress COMMAND quant_price_live --stress-test)
    set_tests_properties(gui_smoke_test gui_tools_test gui_laboratory_test gui_visuals_test gui_privacy_and_lifecycle bounded_history_stress PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
  endif()
endif()
if(BUILD_BENCHMARKS)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(benchmark)
  add_executable(quant_benchmark benchmarks/bench_analytics.cpp)
  target_link_libraries(quant_benchmark PRIVATE quant benchmark::benchmark)
endif()
find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
  add_custom_target(docs COMMAND "${DOXYGEN_EXECUTABLE}" Doxyfile
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" COMMENT "Generate Doxygen documentation")
endif()
